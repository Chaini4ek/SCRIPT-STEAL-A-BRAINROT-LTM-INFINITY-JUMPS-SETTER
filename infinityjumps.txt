local Players = game:GetService("Players")
local UserInputService = game:GetService("UserInputService")
local LocalPlayer = Players.LocalPlayer

local currentJumps = 999

local guiParent = (gethui and gethui()) or game:GetService("CoreGui")
local gui = Instance.new("ScreenGui")
gui.Name = "JumpSetter"
gui.ResetOnSpawn = false
gui.Parent = guiParent

local frame = Instance.new("Frame")
frame.Size = UDim2.new(0, 220, 0, 110)
frame.Position = UDim2.new(0, 20, 0, 100)
frame.BackgroundColor3 = Color3.fromRGB(18, 18, 22)
frame.BorderSizePixel = 0
frame.Active = true
frame.Draggable = true
frame.Parent = gui

local corner = Instance.new("UICorner")
corner.CornerRadius = UDim.new(0, 10)
corner.Parent = frame

local stroke = Instance.new("UIStroke")
stroke.Color = Color3.fromRGB(123, 253, 70)
stroke.Thickness = 1.5
stroke.Parent = frame

local title = Instance.new("TextLabel")
title.Size = UDim2.new(1, 0, 0, 28)
title.BackgroundColor3 = Color3.fromRGB(28, 28, 34)
title.Text = "Jumps"
title.TextColor3 = Color3.fromRGB(123, 253, 70)
title.Font = Enum.Font.GothamBold
title.TextSize = 15
title.Parent = frame
local tc = Instance.new("UICorner")
tc.CornerRadius = UDim.new(0, 10)
tc.Parent = title

local box = Instance.new("TextBox")
box.Size = UDim2.new(1, -20, 0, 32)
box.Position = UDim2.new(0, 10, 0, 36)
box.BackgroundColor3 = Color3.fromRGB(40, 40, 46)
box.Text = "999"

box.TextColor3 = Color3.fromRGB(230, 230, 230)

box.TextSize = 15

box.Parent = frame
local bc = Instance.new("UICorner")
bc.CornerRadius = UDim.new(0, 6)
bc.Parent = box

local btn = Instance.new("TextButton")
btn.Size = UDim2.new(1, -20, 0, 32)
btn.Position = UDim2.new(0, 10, 0, 72)
btn.BackgroundColor3 = Color3.fromRGB(60, 160, 80)
btn.Text = "Set"
btn.TextColor3 = Color3.fromRGB(255, 255, 255)
btn.Font = Enum.Font.GothamBold
btn.TextSize = 15
btn.Parent = frame
local c = Instance.new("UICorner")
c.CornerRadius = UDim.new(0, 6)
c.Parent = btn

local function applyJump()
    local n = tonumber(box.Text)
    if not n then return end
    if n < 0 then n = 0 end
    if n > 100000 then n = 100000 end
    currentJumps = math.floor(n)
    pcall(function()
        LocalPlayer:SetAttribute("ExtraJumps", currentJumps)
    end)
end

btn.MouseButton1Click:Connect(applyJump)
box.FocusLost:Connect(applyJump)

task.spawn(function()
    while true do
        task.wait(1)
        pcall(function()
            local cur = LocalPlayer:GetAttribute("ExtraJumps") or 0
            if cur < currentJumps then
                LocalPlayer:SetAttribute("ExtraJumps", currentJumps)
            end
        end)
    end
end)

LocalPlayer.CharacterAdded:Connect(function()
    task.wait(0.5)
    applyJump()
end)

applyJump()
